COMMENT UTILISER:
─────────────────────────────────────────────────────────────────────────────

Exemple minimal:

  #include "fem_2d.hpp"
  
  // Créer le problème
  ElasticityFEM2D fem(width=10.0, height=5.0, 
                       nx=31, ny=16, E=200e9, nu=0.3);
  
  // Conditions aux limites
  std::map<std::string, std::vector<std::pair<int,double>>> bcs;
  bcs["bottom"] = {{1, 0.0}};
  bcs["top"] = {{1, 0.02}};
  bcs["left"] = {{0, 0.0}};
  
  // Résoudre
  fem.solve(bcs);
  fem.compute_stress();
  
  // Résultats
  SVGGenerator::write_von_mises(fem, "result.svg");
